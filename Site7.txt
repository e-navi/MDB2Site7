    /// <summary>
    /// 3次元スプライン曲線を計算するクラス
    /// </summary>
    public class Xross_Spline {
        private int N;
        private double[] _parameterP;

        /// <summary>
        /// スプライン補間用のテーブルを作成する
        /// </summary>
        private void CurveMakeTable(double[] x, double[] y, ref double[] z) {
            int n = x.Length;
            double[] h = new double[n];
            double[] d = new double[n];
            double t;

            z[0] = 0;
            z[n - 1] = 0;

            for (int i = 0; i < n - 1; i++) {
                // 前後のパラメータが同じ場合は微小移動して計算不能を避ける
                if (Math.Abs(x[i + 1] - x[i]) < 1E-12) {
                    x[i + 1] += 1E-08;
                }
                h[i] = x[i + 1] - x[i];
                d[i + 1] = (y[i + 1] - y[i]) / h[i];
            }

            z[1] = d[2] - d[1] - h[0] * z[0];
            d[1] = 2 * (x[2] - x[0]);

            for (int i = 1; i < n - 2; i++) {
                t = h[i] / d[i];
                z[i + 1] = d[i + 2] - d[i + 1] - z[i] * t;
                d[i + 1] = 2 * (x[i + 2] - x[i]) - h[i] * t;
            }

            z[n - 2] -= h[n - 2] * z[n - 1];

            for (int i = n - 2; i > 0; i--) {
                z[i] = (z[i] - h[i] * z[i + 1]) / d[i];
            }
        }

        /// <summary>
        /// 閉曲線用の補間テーブルを作成する
        /// </summary>
        private void CloseCurveMakeTable(double[] x, double[] y, ref double[] z) {
            int n = x.Length - 1;
            double[] h = new double[n + 1];
            double[] d = new double[n + 1];
            double[] w = new double[n + 1];
            double t;

            for (int i = 0; i < n; i++) {
                if (Math.Abs(x[i + 1] - x[i]) < 1E-12) {
                    x[i + 1] += 1E-08;
                }
                h[i] = x[i + 1] - x[i];
                w[i] = (y[i + 1] - y[i]) / h[i];
            }

            w[n] = w[0];

            for (int i = 1; i < n; i++) {
                d[i] = 2 * (x[i + 1] - x[i - 1]);
            }
            d[n] = 2 * (h[n - 1] + h[0]);

            for (int i = 1; i <= n; i++) {
                z[i] = w[i] - w[i - 1];
            }

            w[1] = h[0];
            w[n - 1] = h[n - 1];
            w[n] = d[n];

            for (int i = 2; i < n - 1; i++)
                w[i] = 0;

            for (int i = 1; i < n; i++) {
                t = h[i] / d[i];
                z[i + 1] -= z[i] * t;
                d[i + 1] -= h[i] * t;
                w[i + 1] -= w[i] * t;
            }

            w[0] = w[n];
            z[0] = z[n];

            for (int i = n - 2; i >= 0; i--) {
                t = h[i] / d[i + 1];
                z[i] -= z[i + 1] * t;
                w[i] -= w[i + 1] * t;
            }

            t = z[0] / w[0];
            z[0] = t;
            z[n] = t;

            for (int i = 1; i < n; i++) {
                z[i] = (z[i] - w[i] * t) / d[i];
            }
        }

        /// <summary>
        /// 任意点 t におけるスプライン値を計算する
        /// </summary>
        public double GetSplineValue(double t, double[] x, double[] y, double[] z) {
            int n = x.Length;
            int i = 0;
            int j = n - 1;

            while (i < j) {
                int k = (i + j) / 2;
                if (x[k] < t)
                    i = k + 1;
                else
                    j = k;
            }

            if (i > 0)
                i--;

            double h = x[i + 1] - x[i];
            double d = t - x[i];

            return (((z[i + 1] - z[i]) * d / h + z[i] * 3) * d + ((y[i + 1] - y[i]) / h - (z[i] * 2 + z[i + 1]) * h)) * d + y[i];
        }

        /// <summary>
        /// 閉曲線における任意点 t のスプライン値を計算する
        /// </summary>
        public double GetCloseSplineValue(double t, double[] x, double[] y, double[] z) {
            int n = x.Length - 1;
            double period = x[n] - x[0];
            while (t > x[n])
                t -= period;
            while (t < x[0])
                t += period;

            int i = 0;
            int j = n;

            while (i < j) {
                int k = (i + j) / 2;
                if (x[k] < t)
                    i = k + 1;
                else
                    j = k;
            }

            if (i > 0)
                i--;

            double h = x[i + 1] - x[i];
            double d = t - x[i];

            return (((z[i + 1] - z[i]) * d / h + z[i] * 3) * d + ((y[i + 1] - y[i]) / h - (z[i] * 2 + z[i + 1]) * h)) * d + y[i];
        }

        /// <summary>
        /// 3D曲線の点群を計算する（開曲線）
        /// </summary>
        public List<XYZ> Calc3DCurvePoints(List<IkouPRec> points, int divideStep) {
            if (points == null || points.Count < 2)
                return new List<XYZ>();

            int n = points.Count;
            double[] x = new double[n];
            double[] y = new double[n];
            double[] z = new double[n];
            _parameterP = new double[n];

            for (int i = 0; i < n; i++) {
                x[i] = points[i].X;
                y[i] = points[i].Y;
                z[i] = points[i].Z;
            }

            // 弦長パラメトリック補間のためのパラメータ計算
            _parameterP[0] = 0;
            for (int i = 1; i < n; i++) {
                double dx = x[i] - x[i - 1];
                double dy = y[i] - y[i - 1];
                double dz = z[i] - z[i - 1];
                _parameterP[i] = _parameterP[i - 1] + Math.Sqrt(dx * dx + dy * dy + dz * dz);
            }

            // パラメータを 0.0 ~ 1.0 に正規化
            double totalLength = _parameterP[n - 1];
            if (totalLength > 0) {
                for (int i = 0; i < n; i++)
                    _parameterP[i] /= totalLength;
            }

            double[] tx = new double[n];
            double[] ty = new double[n];
            double[] tz = new double[n];

            CurveMakeTable(_parameterP, x, ref tx);
            CurveMakeTable(_parameterP, y, ref ty);
            CurveMakeTable(_parameterP, z, ref tz);

            List<XYZ> result = new List<XYZ>();
            int createCount = (n - 1) * divideStep;
            for (int i = 0; i <= createCount; i++) {
                double t = (double)i / createCount;
                double px = GetSplineValue(t, _parameterP, x, tx);
                double py = GetSplineValue(t, _parameterP, y, ty);
                double pz = GetSplineValue(t, _parameterP, z, tz);
                result.Add(new XYZ(px, py, pz));
            }

            return result;
        }

        /// <summary>
        /// 3D曲線の点群を計算する（閉曲線）
        /// </summary>
        public List<XYZ> Calc3DCloseCurvePoints(List<IkouPRec> points, int divideStep) {
            if (points == null || points.Count < 3)
                return new List<XYZ>();

            int n = points.Count - 1;
            double[] x = new double[points.Count];
            double[] y = new double[points.Count];
            double[] z = new double[points.Count];
            _parameterP = new double[points.Count];

            for (int i = 0; i < points.Count; i++) {
                x[i] = points[i].X;
                y[i] = points[i].Y;
                z[i] = points[i].Z;
            }

            _parameterP[0] = 0;
            for (int i = 1; i <= n; i++) {
                double dx = x[i] - x[i - 1];
                double dy = y[i] - y[i - 1];
                double dz = z[i] - z[i - 1];
                _parameterP[i] = _parameterP[i - 1] + Math.Sqrt(dx * dx + dy * dy + dz * dz);
            }

            double totalLength = _parameterP[n];
            if (totalLength > 0) {
                for (int i = 0; i <= n; i++)
                    _parameterP[i] /= totalLength;
            }

            double[] tx = new double[n + 1];
            double[] ty = new double[n + 1];
            double[] tz = new double[n + 1];

            CloseCurveMakeTable(_parameterP, x, ref tx);
            CloseCurveMakeTable(_parameterP, y, ref ty);
            CloseCurveMakeTable(_parameterP, z, ref tz);

            List<XYZ> result = new List<XYZ>();
            int createCount = n * divideStep;
            for (int i = 0; i <= createCount; i++) {
                double t = (double)i / createCount;
                double px = GetCloseSplineValue(t, _parameterP, x, tx);
                double py = GetCloseSplineValue(t, _parameterP, y, ty);
                double pz = GetCloseSplineValue(t, _parameterP, z, tz);
                result.Add(new XYZ(px, py, pz));
            }

            return result;
        }
    }

■遺構図面の作成について

真上が真北の全図から遺構を切り出して、遺構図面へ配置する。
単一遺構と遺構範囲(複数遺構)の2種類ある。
　　単一遺構は、遺構を指定してそのデータのみが対象となる
　　遺構範囲は、その範囲に含まれるすべての遺構が対象となる。
3点を支持して切り出すのはどちらも同じ。
1点目は左下、2点目は右下、3点目は上方向。
これにより矩形と回転角が得られる。
方位は、回転角から求める。

        public List<Point3D> CalcDanmenP3() {
            double dx = (ep.X - sp.X) / cnt;
            double dy = (ep.Y - sp.Y) / cnt;
            List<Point3D> list = new List<Point3D>();

            for (int i = 0; i < cnt; i++) {
                list.Add(new Point3D(sp.X + dx * i, sp.Y + dy * i, danmen[i].Elevation));
            }
            return list;
        }
        public List<Point3D> CalcDanmenP2() {
            double dx = (ep.X - sp.X) / cnt;
            double dy = (ep.Y - sp.Y) / cnt;
            double len = Math.Sqrt(Math.Pow(ep.X - sp.X, 2) + Math.Pow(ep.Y - sp.Y, 2));
            double dx2 = (ep.X - sp.X) / len;
            double dy2 = (ep.Y - sp.Y) / len;
            List<Point3D> list = new List<Point3D>();

            double se = danmen[0].Elevation;
            for (int i = 0; i < cnt; i++) {
                double e = danmen[i].Elevation - se;
                double x = dp.X + dx * i - dy2 * e;
                double y = dp.Y + dy * i + dx2 * e;

                list.Add(new Point3D(x, y, 0));
            }
            return list;
        }
    }

        public static List<Point3D> LoadPointsFromTsv(string filePath)
        {
            var points = new List<Point3D>();
            if (!File.Exists(filePath)) return points;

            var lines = File.ReadAllLines(filePath);
            int startLine = (lines.Length > 0 && lines[0].Contains("X")) ? 1 : 0;

            for (int i = startLine; i < lines.Length; i++)
            {
                var parts = lines[i].Split('\t');
                if (parts.Length >= 3 &&
                    double.TryParse(parts[0], NumberStyles.Any, CultureInfo.InvariantCulture, out double x) &&
                    double.TryParse(parts[1], NumberStyles.Any, CultureInfo.InvariantCulture, out double y) &&
                    double.TryParse(parts[2], NumberStyles.Any, CultureInfo.InvariantCulture, out double z))
                {
                    points.Add(new Point3D(x, y, z));
                }
            }
            return points;
        }

        {

            mesh.ResolutionX = resolutionX;
            mesh.ResolutionY = resolutionY;
            if (sourcePoints.Count == 0) return mesh;

            /*
            double minX = sourcePoints.Min(p => p.X);
            double maxX = sourcePoints.Max(p => p.X);
            double minY = sourcePoints.Min(p => p.Y);
            double maxY = sourcePoints.Max(p => p.Y);

            //追加
            double dX = maxX - minX;
            double dY = maxY - minY;
            minX -= dX * 0.2;
            minY -= dY * 0.2;
            maxX += dX * 0.2;
            maxY += dY * 0.2;
            */
            double minX = minp.X;
            double maxX = maxp.X;
            double minY = minp.Y;
            double maxY = maxp.Y;


            double stepX = (maxX - minX) / (resolutionX - 1);
            double stepY = (maxY - minY) / (resolutionY - 1);

            // Calculate a heuristic smoothing factor (e.g. 5% of average step)
            double smoothing = Math.Sqrt(stepX * stepX + stepY * stepY) * 0.5;

            for (int j = 0; j < resolutionY; j++)
            {
                for (int i = 0; i < resolutionX; i++)
                {
                    double x = minX + i * stepX;
                    double y = minY + j * stepY;
                    double z = CalculateIdw(new Point3D(x, y, 0), sourcePoints, power, smoothing);
                    mesh.Positions.Add(new Point3D(x, y, z));
                }
            }

            for (int j = 0; j < resolutionY - 1; j++)
            {
                for (int i = 0; i < resolutionX - 1; i++)
                {
                    int i0 = j * resolutionX + i;
                    int i1 = i0 + 1;
                    int i2 = i0 + resolutionX;
                    int i3 = i2 + 1;

                    // Counter-clockwise
                    mesh.TriangleIndices.Add(i0);
                    mesh.TriangleIndices.Add(i2);
                    mesh.TriangleIndices.Add(i1);

                    mesh.TriangleIndices.Add(i1);
                    mesh.TriangleIndices.Add(i2);
                    mesh.TriangleIndices.Add(i3);
                }
            }

            return mesh;
        }

        private static double CalculateIdw(Point3D target, List<Point3D> points, double power = 2.0, double smoothing = 0.0)
        {
            double sumWeights = 0;
            double sumWeightedValues = 0;

            foreach (var p in points)
            {
                double dx = target.X - p.X;
                double dy = target.Y - p.Y;
                double distSq = dx * dx + dy * dy;
                
                // If smoothing is > 0, we use (dist^2 + smoothing^2)
                // This prevents the weight from going to infinity and smooths peaks
                double d = Math.Sqrt(distSq + smoothing * smoothing);
                
                if (d < 0.0001) return p.Z;

                double weight = 1.0 / Math.Pow(d, power);
                sumWeights += weight;
                sumWeightedValues += weight * p.Z;
            }

            return sumWeightedValues / (sumWeights + 1e-12);
        }
        public static List<(double Distance, double Elevation)> SampleProfile(
            Point3D start,
            Point3D end,
            List<Point3D> sourcePoints,
            int sampleCount = 100,
            double power = 2.0) {
            var profile = new List<(double, double)>();
            if (sourcePoints.Count == 0)
                return profile;

            double dx = end.X - start.X;
            double dy = end.Y - start.Y;
            double totalDist = Math.Sqrt(dx * dx + dy * dy);

            // Heuristic smoothing for profile
            double smoothing = totalDist / (sampleCount > 1 ? sampleCount - 1 : 1) * 0.5;

            for (int i = 0; i < sampleCount; i++) {
                double t = (double)i / (sampleCount - 1);
                double x = start.X + dx * t;
                double y = start.Y + dy * t;
                double z = CalculateIdw(new Point3D(x, y, 0), sourcePoints, power, smoothing);
                profile.Add((t * totalDist, z));
            }

            return profile;
        }
        public static Danmen CalcDanmen(
            Point3D start,
            Point3D end,
            Point3D dp,
            List<Point3D> sourcePoints,
            int cnt = 100,
            double power = 5.0) {
            var danmen = new Danmen();
            if (sourcePoints.Count == 0)
                return danmen;

            double dx = end.X - start.X;
            double dy = end.Y - start.Y;
            double totalDist = Math.Sqrt(dx * dx + dy * dy);

            // Heuristic smoothing for profile
            double smoothing = totalDist / (cnt > 1 ? cnt - 1 : 1) * 0.5;

            for (int i = 0; i < cnt; i++) {
                double t = (double)i / (cnt - 1);
                double x = start.X + dx * t;
                double y = start.Y + dy * t;
                double z = CalculateIdw(new Point3D(x, y, 0), sourcePoints, power, smoothing);
                danmen.danmen.Add((t * totalDist, z));
            }
            danmen.sp = start;
            danmen.ep = end;
            danmen.dp = dp;
            danmen.cnt = cnt;
            return danmen;
        }
